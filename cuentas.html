DROP FUNCTION IF EXISTS register_payment(uuid, uuid, uuid, numeric); CREATE OR REPLACE FUNCTION register_payment( p_business_id uuid, p_sale_id uuid, p_payment_method_id uuid, p_amount numeric ) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE v_sale record; v_new_balance numeric; v_new_status text; v_payment_id uuid; BEGIN /* ========================================= VALIDAR MONTO ========================================= */ IF p_amount IS NULL OR p_amount <= 0 THEN RAISE EXCEPTION 'El monto del abono debe ser mayor a 0.'; END IF; /* ========================================= BUSCAR LA VENTA ========================================= */ SELECT id, business_id, customer_id, total, balance, status INTO v_sale FROM sales WHERE id = p_sale_id FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'La venta no existe.'; END IF; /* ========================================= VALIDAR NEGOCIO ========================================= */ IF v_sale.business_id IS DISTINCT FROM p_business_id THEN RAISE EXCEPTION 'La venta no pertenece al negocio seleccionado.'; END IF; /* ========================================= VALIDAR SALDO ========================================= */ IF COALESCE(v_sale.balance, 0) <= 0 THEN RAISE EXCEPTION 'Esta venta ya no tiene saldo pendiente.'; END IF; IF p_amount > v_sale.balance THEN RAISE EXCEPTION 'El abono ($%) no puede ser mayor al saldo pendiente ($%).', p_amount, v_sale.balance; END IF; /* ========================================= REGISTRAR EL ABONO IMPORTANTE: business_id se guarda explícitamente. ========================================= */ INSERT INTO payments ( business_id, sale_id, payment_method_id, amount, paid_at ) VALUES ( p_business_id, p_sale_id, p_payment_method_id, p_amount, NOW() ) RETURNING id INTO v_payment_id; /* ========================================= CALCULAR NUEVO SALDO ========================================= */ v_new_balance = ROUND( GREATEST( COALESCE(v_sale.balance, 0) - p_amount, 0 ), 2 ); /* ========================================= DETERMINAR ESTADO $0.00 = paid > $0.00 = open ========================================= */ IF v_new_balance <= 0 THEN v_new_balance := 0; v_new_status := 'paid'; ELSE v_new_status := 'open'; END IF; /* ========================================= ACTUALIZAR VENTA ========================================= */ UPDATE sales SET balance = v_new_balance, status = v_new_status WHERE id = p_sale_id AND business_id = p_business_id; /* ========================================= DEVOLVER RESULTADO ========================================= */ RETURN jsonb_build_object( 'success', true, 'payment_id', v_payment_id, 'sale_id', p_sale_id, 'business_id', p_business_id, 'amount', p_amount, 'previous_balance', v_sale.balance, 'new_balance', v_new_balance, 'status', v_new_status ); END; $$;
